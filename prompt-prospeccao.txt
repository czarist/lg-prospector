# === Prospecção (contínua) — NÃO dispara e-mail ===

# Bater a cota N não encerra o nicho×cidade — a próxima volta muda a query.
# --skip-completed volta ao modo escada (uma vez por par).
# --max-partial-attempts 5 só conta tentativa fraca; sem --skip-completed não para.
# Disparo: scripts/mailman.py (em paralelo).

# --- HUNT (escada atual) ---
cd /home/lucas/lg-prospector && source .venv/bin/activate && \

  --focus-rs \
  --max-tier 3 \
  --min-pop-k 50 \
  -n 10 \
  --stages discover,enrich,crm \
  --pause 8 \
  --cycle-pause 60 \
  --max-partial-attempts 5
